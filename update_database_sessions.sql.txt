USE bdd_uka;

-- ----------------------------------------------------------------------------
-- Table: sessions_actives
-- Description: Stocke toutes les sessions actives des utilisateurs
--              Une seule session active par utilisateur à la fois
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS `sessions_actives` (
  `id_session` INT NOT NULL AUTO_INCREMENT,
  `session_id` VARCHAR(255) NOT NULL COMMENT 'ID unique de la session PHP',
  `user_login` VARCHAR(45) CHARACTER SET utf8mb3 COLLATE utf8mb3_general_ci NOT NULL COMMENT 'Login de l\'utilisateur',
  `user_matricule` VARCHAR(20) CHARACTER SET utf8mb3 COLLATE utf8mb3_general_ci NOT NULL COMMENT 'Matricule de l\'agent',
  `type_compte` ENUM('agent', 'jury') NOT NULL DEFAULT 'agent' COMMENT 'Type de compte connecté',
  `ip_address` VARCHAR(45) DEFAULT NULL COMMENT 'Adresse IP de connexion',
  `user_agent` TEXT DEFAULT NULL COMMENT 'Navigateur et système utilisé',
  `date_connexion` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Date et heure de connexion',
  `derniere_activite` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT 'Dernière activité détectée',
  `statut` ENUM('active', 'inactive', 'expiration_en_cours') NOT NULL DEFAULT 'active' COMMENT 'État de la session',
  `token_validation` VARCHAR(64) DEFAULT NULL COMMENT 'Token unique pour validation',
  PRIMARY KEY (`id_session`),
  UNIQUE KEY `unique_session_id` (`session_id`),
  KEY `idx_user_login` (`user_login`),
  KEY `idx_matricule` (`user_matricule`),
  KEY `idx_statut` (`statut`),
  KEY `idx_derniere_activite` (`derniere_activite`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci
COMMENT='Gestion des sessions actives avec un seul utilisateur connecté à la fois';

-- ----------------------------------------------------------------------------
-- Table: demandes_connexion
-- Description: File d'attente pour les tentatives de connexion concurrentes
--              Permet de notifier l'utilisateur actuel et gérer les validations
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS `demandes_connexion` (
  `id_demande` INT NOT NULL AUTO_INCREMENT,
  `user_login` VARCHAR(45) CHARACTER SET utf8mb3 COLLATE utf8mb3_general_ci NOT NULL COMMENT 'Login de l\'utilisateur demandant la connexion',
  `user_matricule` VARCHAR(20) CHARACTER SET utf8mb3 COLLATE utf8mb3_general_ci NOT NULL COMMENT 'Matricule de l\'agent',
  `type_compte` ENUM('agent', 'jury') NOT NULL DEFAULT 'agent' COMMENT 'Type de compte à connecter',
  `ip_address` VARCHAR(45) DEFAULT NULL COMMENT 'IP de la nouvelle demande',
  `user_agent` TEXT DEFAULT NULL COMMENT 'Navigateur de la nouvelle demande',
  `date_demande` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Date de la tentative de connexion',
  `statut_demande` ENUM('en_attente', 'acceptee', 'refusee', 'expiree') NOT NULL DEFAULT 'en_attente' COMMENT 'État de la demande',
  `reponse_user_actuel` ENUM('en_attente', 'accepte', 'refuse') DEFAULT 'en_attente' COMMENT 'Réponse de l\'utilisateur actuellement connecté',
  `date_reponse_actuel` DATETIME DEFAULT NULL COMMENT 'Date de réponse de l\'utilisateur actuel',
  `reponse_user_demandeur` ENUM('en_attente', 'accepte', 'refuse') DEFAULT 'en_attente' COMMENT 'Confirmation du demandeur',
  `date_reponse_demandeur` DATETIME DEFAULT NULL COMMENT 'Date de confirmation du demandeur',
  `session_id_actuelle` VARCHAR(255) DEFAULT NULL COMMENT 'ID de la session à remplacer',
  `token_demande` VARCHAR(64) NOT NULL COMMENT 'Token unique pour cette demande',
  `expiration` DATETIME DEFAULT NULL COMMENT 'Date d\'expiration de la demande',
  PRIMARY KEY (`id_demande`),
  UNIQUE KEY `unique_token_demande` (`token_demande`),
  KEY `idx_user_login` (`user_login`),
  KEY `idx_statut_demande` (`statut_demande`),
  KEY `idx_expiration` (`expiration`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci
COMMENT='File d\'attente des demandes de connexion avec validation bidirectionnelle';

-- ----------------------------------------------------------------------------
-- Table: historique_sessions
-- Description: Archive de toutes les sessions (active ou terminée)
--              Permet de tracer l'activité et les connexions multiples
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS `historique_sessions` (
  `id_historique` INT NOT NULL AUTO_INCREMENT,
  `session_id` VARCHAR(255) NOT NULL COMMENT 'ID de la session PHP',
  `user_login` VARCHAR(45) CHARACTER SET utf8mb3 COLLATE utf8mb3_general_ci NOT NULL COMMENT 'Login de l\'utilisateur',
  `user_matricule` VARCHAR(20) CHARACTER SET utf8mb3 COLLATE utf8mb3_general_ci NOT NULL COMMENT 'Matricule de l\'agent',
  `type_compte` ENUM('agent', 'jury') NOT NULL DEFAULT 'agent' COMMENT 'Type de compte',
  `ip_address` VARCHAR(45) DEFAULT NULL COMMENT 'Adresse IP',
  `user_agent` TEXT DEFAULT NULL COMMENT 'Navigateur utilisé',
  `date_connexion` DATETIME NOT NULL COMMENT 'Date de connexion',
  `date_deconnexion` DATETIME DEFAULT NULL COMMENT 'Date de déconnexion',
  `duree_session` INT DEFAULT NULL COMMENT 'Durée en secondes',
  `raison_deconnexion` ENUM('manuelle', 'inactivite', 'force', 'concurrent', 'expiration') DEFAULT NULL COMMENT 'Raison de la déconnexion',
  `nombre_activites` INT DEFAULT 0 COMMENT 'Nombre d\'activités durant la session',
  PRIMARY KEY (`id_historique`),
  KEY `idx_user_login` (`user_login`),
  KEY `idx_matricule` (`user_matricule`),
  KEY `idx_date_connexion` (`date_connexion`),
  KEY `idx_raison` (`raison_deconnexion`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci
COMMENT='Historique complet de toutes les sessions pour audit et statistiques';

-- ----------------------------------------------------------------------------
-- Table: activites_session
-- Description: Log des activités pendant une session (optionnel)
--              Permet de suivre l'activité détaillée de l'utilisateur
-- ----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS `activites_session` (
  `id_activite` INT NOT NULL AUTO_INCREMENT,
  `session_id` VARCHAR(255) NOT NULL COMMENT 'ID de la session PHP',
  `user_login` VARCHAR(45) CHARACTER SET utf8mb3 COLLATE utf8mb3_general_ci NOT NULL COMMENT 'Login de l\'utilisateur',
  `page_visitee` VARCHAR(255) DEFAULT NULL COMMENT 'URL ou page visitée',
  `action` VARCHAR(100) DEFAULT NULL COMMENT 'Type d\'action (clic, navigation, etc.)',
  `date_activite` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Timestamp de l\'activité',
  `donnees_supplementaires` TEXT DEFAULT NULL COMMENT 'Données JSON supplémentaires',
  PRIMARY KEY (`id_activite`),
  KEY `idx_session_id` (`session_id`),
  KEY `idx_date_activite` (`date_activite`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci
COMMENT='Log détaillé des activités par session (optionnel pour analytics)';

-- ----------------------------------------------------------------------------
-- Procédure stockée: Nettoyer les sessions inactives
-- Description: Supprime automatiquement les sessions inactives après X minutes
-- ----------------------------------------------------------------------------
DELIMITER $$

DROP PROCEDURE IF EXISTS `clean_inactive_sessions`$$

CREATE PROCEDURE `clean_inactive_sessions`(IN inactivity_minutes INT)
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE v_session_id VARCHAR(255);
    DECLARE v_user_login VARCHAR(45);
    DECLARE v_date_connexion DATETIME;
    DECLARE v_derniere_activite DATETIME;
    
    -- Curseur pour parcourir les sessions inactives
    DECLARE cur_sessions CURSOR FOR 
        SELECT session_id, user_login, date_connexion, derniere_activite
        FROM sessions_actives 
        WHERE statut = 'active' 
        AND derniere_activite < DATE_SUB(NOW(), INTERVAL inactivity_minutes MINUTE);
    
    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;
    
    OPEN cur_sessions;
    
    read_loop: LOOP
        FETCH cur_sessions INTO v_session_id, v_user_login, v_date_connexion, v_derniere_activite;
        
        IF done THEN
            LEAVE read_loop;
        END IF;
        
        -- Archiver dans l'historique
        INSERT INTO historique_sessions 
            (session_id, user_login, user_matricule, type_compte, ip_address, user_agent, 
             date_connexion, date_deconnexion, duree_session, raison_deconnexion)
        SELECT 
            session_id, user_login, user_matricule, type_compte, ip_address, user_agent,
            date_connexion, NOW(), 
            TIMESTAMPDIFF(SECOND, date_connexion, NOW()),
            'inactivite'
        FROM sessions_actives 
        WHERE session_id = v_session_id;
        
        -- Supprimer la session active
        DELETE FROM sessions_actives WHERE session_id = v_session_id;
        
        -- Mettre à jour les tables compte_agent ou t_membre_jury
        UPDATE compte_agent 
        SET session_active = 0, session_id_actuelle = NULL 
        WHERE session_id_actuelle = v_session_id;
        
        UPDATE t_membre_jury 
        SET session_active = 0, session_id_actuelle = NULL 
        WHERE session_id_actuelle = v_session_id;
        
    END LOOP;
    
    CLOSE cur_sessions;
    
    -- Nettoyer aussi les demandes expirées
    UPDATE demandes_connexion 
    SET statut_demande = 'expiree' 
    WHERE statut_demande = 'en_attente' 
    AND expiration < NOW();
    
END$$

DELIMITER ;

-- ----------------------------------------------------------------------------
-- Event scheduler: Nettoyer automatiquement toutes les 5 minutes
-- Description: Active le nettoyage automatique des sessions inactives
-- ----------------------------------------------------------------------------

-- Activer l'event scheduler si pas déjà fait
SET GLOBAL event_scheduler = ON;

-- Créer l'événement de nettoyage automatique
DROP EVENT IF EXISTS `auto_clean_sessions`;

CREATE EVENT `auto_clean_sessions`
ON SCHEDULE EVERY 5 MINUTE
STARTS CURRENT_TIMESTAMP
DO
    CALL clean_inactive_sessions(30); -- Nettoyer sessions inactives depuis 30 minutes

-- Vue: Sessions actives avec informations utilisateur
CREATE OR REPLACE VIEW `v_sessions_actives` AS
SELECT 
    sa.id_session,
    sa.session_id,
    sa.user_login,
    sa.user_matricule,
    sa.type_compte,
    CASE 
        WHEN sa.type_compte = 'agent' THEN CONCAT(a.Nom_agent, ' ', a.Post_agent, ' ', a.Prenom)
        ELSE CONCAT(aj.Nom_agent, ' ', aj.Post_agent, ' ', aj.Prenom)
    END AS nom_complet,
    sa.ip_address,
    sa.date_connexion,
    sa.derniere_activite,
    TIMESTAMPDIFF(MINUTE, sa.derniere_activite, NOW()) AS minutes_inactif,
    sa.statut
FROM sessions_actives sa
LEFT JOIN compte_agent ca ON sa.user_login = ca.Login AND sa.type_compte = 'agent'
LEFT JOIN agent a ON ca.Mat_agent = a.Mat_agent
LEFT JOIN t_membre_jury tmj ON sa.user_login = tmj.Login AND sa.type_compte = 'jury'
LEFT JOIN agent aj ON tmj.Mat_agent = aj.Mat_agent;

-- Vue: Demandes de connexion en attente avec détails
CREATE OR REPLACE VIEW `v_demandes_connexion_actives` AS
SELECT 
    dc.id_demande,
    dc.user_login,
    dc.user_matricule,
    CASE 
        WHEN dc.type_compte = 'agent' THEN CONCAT(a.Nom_agent, ' ', a.Post_agent, ' ', a.Prenom)
        ELSE CONCAT(aj.Nom_agent, ' ', aj.Post_agent, ' ', aj.Prenom)
    END AS nom_complet,
    dc.type_compte,
    dc.ip_address,
    dc.date_demande,
    dc.statut_demande,
    dc.reponse_user_actuel,
    dc.reponse_user_demandeur,
    dc.token_demande,
    TIMESTAMPDIFF(MINUTE, dc.date_demande, NOW()) AS minutes_attente
FROM demandes_connexion dc
LEFT JOIN compte_agent ca ON dc.user_login = ca.Login AND dc.type_compte = 'agent'
LEFT JOIN agent a ON ca.Mat_agent = a.Mat_agent
LEFT JOIN t_membre_jury tmj ON dc.user_login = tmj.Login AND dc.type_compte = 'jury'
LEFT JOIN agent aj ON tmj.Mat_agent = aj.Mat_agent
WHERE dc.statut_demande = 'en_attente';
