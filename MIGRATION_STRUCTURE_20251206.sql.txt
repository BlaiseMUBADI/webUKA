DROP PROCEDURE IF EXISTS `Ajouter_Membre_Jury`;

DELIMITER $$
CREATE DEFINER=`blaise`@`%` PROCEDURE `Ajouter_Membre_Jury`(
    IN p_id_jury INT,
    IN p_mat_agent VARCHAR(50),
    IN p_role ENUM('Président','Secrétaire','Membre'),
    IN p_login VARCHAR(50),
    IN p_mot_passe VARCHAR(255),
    IN p_statut ENUM('Actif', 'Inactif'),
    OUT p_success BOOLEAN,
    OUT p_message VARCHAR(255),
    OUT p_id_membre INT
)
BEGIN
    DECLARE v_count INT DEFAULT 0;
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        SET p_success = FALSE;
        SET p_message = 'Erreur lors de l\'ajout du membre au jury.';
        SET p_id_membre = NULL;
        ROLLBACK;
    END;

    START TRANSACTION;

    -- Initialiser les valeurs de sortie
    SET p_success = FALSE;
    SET p_message = '';
    SET p_id_membre = NULL;

    -- Vérifier que toutes les données nécessaires sont présentes
    IF p_id_jury IS NULL OR p_mat_agent IS NULL OR p_role IS NULL THEN
        SET p_message = 'Données manquantes : id_jury, mat_agent et role sont requis.';
        ROLLBACK;
    ELSE
        -- =============================================================================
        -- VÉRIFICATION ANTI-DOUBLON PAR MATRICULE (Mat_agent)
        -- =============================================================================
        -- LOGIQUE DU SYSTÈME D'AUTHENTIFICATION DUAL (voir index.php):
        --
        -- Un même agent (identifié par Mat_agent) peut avoir DEUX types de comptes:
        --   1. Compte administratif dans 'compte_agent' (avec son propre Login/Password)
        --   2. Compte jury dans 't_membre_jury' (avec un Login/Password différent)
        --
        -- RÈGLE IMPORTANTE pour t_membre_jury:
        --   - Un agent peut être membre de PLUSIEURS jurys différents (jury A, jury B, etc.)
        --   - MAIS il ne peut avoir qu'UN SEUL compte par jury spécifique
        --
        -- Exemples:
        --   ✓ Agent matricule '001' → compte_agent (login: 'admin001') + jury A (login: 'jury001a')
        --   ✓ Agent matricule '001' → jury A (login: 'jury001a') + jury B (login: 'jury001b')
        --   ✗ Agent matricule '001' → jury A (login: 'jury001a') + jury A (login: 'jury001b2')
        --
        -- Cette vérification empêche le dernier cas (doublon dans le même jury)
        -- =============================================================================
        
        SELECT COUNT(*) INTO v_count 
        FROM t_membre_jury 
        WHERE ID_jury = p_id_jury AND Mat_agent = p_mat_agent;

        IF v_count > 0 THEN
            SET p_message = CONCAT('L''agent avec le matricule ', p_mat_agent, ' est déjà membre de ce jury (ID_jury=', p_id_jury, '). Un agent ne peut avoir qu''un seul compte par jury.');
            ROLLBACK;
        ELSE
            -- Si Président ou Secrétaire, vérifier que login et password sont fournis
            IF (p_role = 'Président' OR p_role = 'Secrétaire') AND (p_login IS NULL OR p_login = '' OR p_mot_passe IS NULL OR p_mot_passe = '') THEN
                SET p_message = 'Le login et le mot de passe sont obligatoires pour les Présidents et Secrétaires.';
                ROLLBACK;
            ELSE
                -- Vérifier que le login n'est pas déjà utilisé (si fourni)
                IF p_login IS NOT NULL AND p_login != '' THEN
                    SELECT COUNT(*) INTO v_count 
                    FROM t_membre_jury 
                    WHERE Login = p_login;

                    IF v_count > 0 THEN
                        SET p_message = 'Ce login est déjà utilisé par un autre membre.';
                        ROLLBACK;
                    ELSE
                        -- Insérer avec Login et Mot_passe (Président ou Secrétaire)
                        INSERT INTO t_membre_jury 
                        (ID_jury, Mat_agent, role, Login, Mot_passe, Statut, date_ajout) 
                        VALUES 
                        (p_id_jury, p_mat_agent, p_role, p_login, p_mot_passe, p_statut, NOW());

                        SET p_id_membre = LAST_INSERT_ID();
                        SET p_success = TRUE;
                        SET p_message = 'Membre ajouté avec succès au jury.';
                        COMMIT;
                    END IF;
                ELSE
                    -- Insérer sans Login et Mot_passe (Membre simple)
                    INSERT INTO t_membre_jury 
                    (ID_jury, Mat_agent, role, date_ajout) 
                    VALUES 
                    (p_id_jury, p_mat_agent, p_role, NOW());

                    SET p_id_membre = LAST_INSERT_ID();
                    SET p_success = TRUE;
                    SET p_message = 'Membre ajouté avec succès au jury.';
                    COMMIT;
                END IF;
            END IF;
        END IF;
    END IF;
END$$
DELIMITER ;

SELECT '✓ Procédure Ajouter_Membre_Jury créée/mise à jour' AS Statut;
